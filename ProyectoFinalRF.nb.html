1. Definición del problema

El objetivo es construir un modelo de regresión para estimar el precio de venta de viviendas (SalePrice).
La métrica principal de evaluación será RMSE, porque penaliza más los errores grandes y está indicada en las instrucciones del proyecto.

4. Ingeniería de características

Se aplican las técnicas vistas en clase: imputación, transformación de variables, codificación categórica, escalado y creación de nuevas variables.

preparar_datos <- function(df){
  df <- df %>%
    mutate(
      MSSubClass = as.character(MSSubClass),
      HouseAge = YrSold - YearBuilt,
      RemodAge = YrSold - YearRemodAdd,
      TotalSF = TotalBsmtSF + X1stFlrSF + X2ndFlrSF,
      TotalBath = FullBath + 0.5 * HalfBath + BsmtFullBath + 0.5 * BsmtHalfBath,
      HasGarage = ifelse(is.na(GarageArea) | GarageArea == 0, 0, 1),
      HasFireplace = ifelse(Fireplaces == 0, 0, 1),
      HasPool = ifelse(PoolArea == 0, 0, 1)
    )
  return(df)
}

train_fe <- preparar_datos(train)
test_fe  <- preparar_datos(test)

5. Preparación para modelado

y <- train_fe$SalePrice
train_ids <- train_fe$Id
test_ids <- test_fe$Id

x <- train_fe %>% select(-SalePrice, -Id)
test_x <- test_fe %>% select(-Id)

# Unir temporalmente train y test para asegurar mismas columnas dummy
combined <- bind_rows(x, test_x)

cat_cols <- names(combined)[sapply(combined, is.character)]
num_cols <- names(combined)[sapply(combined, is.numeric)]
# Imputación numérica con mediana
for(col in num_cols){
  mediana_col <- median(combined[[col]], na.rm = TRUE)
  combined[[col]][is.na(combined[[col]])] <- mediana_col
}

# Imputación categórica con "None"
for(col in cat_cols){
  combined[[col]][is.na(combined[[col]]) | combined[[col]] == ""] <- "None"
  combined[[col]] <- as.factor(combined[[col]])
}

# One Hot Encoding
encoder <- dummyVars(~ ., data = combined, fullRank = TRUE)
combined_dummy <- as.data.frame(predict(encoder, newdata = combined))

# División nuevamente
x_dummy <- combined_dummy[1:nrow(x), ]
test_dummy <- combined_dummy[(nrow(x) + 1):nrow(combined_dummy), ]

# Transformación logarítmica de la variable objetivo
log_y <- log1p(y)

6. Validación cruzada y modelos

Se entrenan varios modelos de regresión. No se utilizan redes neuronales, según la restricción del proyecto.

set.seed(123)
train_index <- createDataPartition(log_y, p = 0.80, list = FALSE)

y_valid <- log_y[-train_index]

rmse_real <- function(real_log, pred_log){
  real <- expm1(real_log)
  pred <- expm1(pred_log)
  sqrt(mean((real - pred)^2))
}

mae_real <- function(real_log, pred_log){
  real <- expm1(real_log)
  pred <- expm1(pred_log)
  mean(abs(real - pred))
}

El modelo final se selecciona automáticamente con base en el menor RMSE de validación.

mejor_modelo <- resultados$Modelo[1]
mejor_modelo

if(mejor_modelo == "Random Forest"){
  final_model <- randomForest(
    x = x_dummy,
    y = log_y,
    ntree = 600,
    mtry = floor(sqrt(ncol(x_dummy))),
    nodesize = 2,
    importance = TRUE
  )
  pred_test_log <- predict(final_model, newdata = test_dummy)
}

if(mejor_modelo == "Gradient Boosting"){
  final_model <- gbm(
    formula = log_y ~ .,
    data = data.frame(log_y = log_y, x_dummy),
    distribution = "gaussian",
    n.trees = 700,
    interaction.depth = 3,
    shrinkage = 0.03,
    bag.fraction = 0.75,
    n.minobsinnode = 2,
    verbose = FALSE
  )
  pred_test_log <- predict(final_model, newdata = test_dummy, n.trees = 700)
}

if(mejor_modelo == "Elastic Net"){
  final_model <- cv.glmnet(
    x = as.matrix(x_dummy),
    y = log_y,
    alpha = 0.5,
    nfolds = 5
  )
  pred_test_log <- predict(final_model, newx = as.matrix(test_dummy), s = "lambda.min")[,1]
}

if(mejor_modelo == "Regresión Lineal"){
  final_model <- lm(log_y ~ ., data = data.frame(log_y = log_y, x_dummy))
  pred_test_log <- predict(final_model, newdata = test_dummy)
}

submission <- data.frame(

10. Conclusiones

  • El problema corresponde a regresión supervisada, porque la variable objetivo SalePrice es numérica.
  • Se aplicó limpieza, imputación, codificación categórica, transformación logarítmica e ingeniería de características.
  • La métrica principal fue RMSE, conforme a las instrucciones del proyecto.
  • Se compararon modelos lineales y modelos de ensamble.
